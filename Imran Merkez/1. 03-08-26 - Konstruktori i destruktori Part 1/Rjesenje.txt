// POKAZIVACI

// 1. ALOKACIJA ( new ) 

// - ( ) -> jedna vrijednost 

// - [ ] -> vise vrijednosti

// 2. DEREFERENCIRANJE ( * )

// - ( ) -> jedna vrijednost -> uvijek deref. *_ECTS = 0 , _ECTS = 000002051A228600

// - [ ] -> vise vrijednosti -> nikad deref. *_naziv = N , _naziv = Nema predmeta

// 3. DESTRUTKOR / DEALOKACIJA ( delete , nullptr )

// - ( ) -> jedna vrijednost -> delete _ECTS
 
// - [ ] -> vise vrijednosti -> delete[] _naziv;

// this -> pok naseg predmeta 
// *this -> vrijednost naseg predmeta

class Predmet {
public:			  // dft. const.
	char* _naziv; // Nema predmeta // nullptr
	//char _naziv[50];

	int _ocjena;  //      5		   //
	int* _ECTS;	  //      0		   //

	// 1. dft const.

	Predmet() {

		_ocjena = 5;

		_ECTS = new int(0);

		//char dftNaziv[] = "Nema predmeta"; // 14 slo. 
		//string dftNaziv = "Nema predmeta";
		const char* dftNaziv = "Nema predmeta"; // string literal -> read-only mem.

		int vel = strlen(dftNaziv) + 1; // 13 sl. + 1 = 14

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, dftNaziv);

	}

	// 4. user-def constr.

	//Predmet predmet1("Programiranje II", 9, 7); // user-def constr.

	Predmet(const char* naziv, int ocjena , int ects ) {

		_ocjena = ocjena;

		_ECTS = new int(ects);

		int vel = strlen(naziv) + 1;
		_naziv = new char[vel];
		strcpy_s(_naziv, vel, naziv);

	}

	// 5. copy constr.

	//Predmet predmetKopija2(predmet1); // shallow copy -> deep copy -> copy constr. 

	// bez & -> pravimo kopiju
	// sa & -> saljemo original 
	// const -> nesto je kontantno -> ne moze se mijenjati
	// & i const -> posaji riginal ali ga zastiti

	Predmet(const Predmet& predmet) { // napravila kopija predmeta1 -> posalji original 


		//predmet._ocjena = 10;

		_ocjena = predmet._ocjena;

		_ECTS = new int(*predmet._ECTS);

		int vel = strlen(predmet._naziv) + 1;

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, predmet._naziv);

	}

	// 6. operator = 

	// predmetOperatorDodjele = predmet2; // shallow copy - deep copy -> operator = 

	Predmet& operator = (const Predmet& predmet) {


		// this -> predmetOperatorDodjele
		// _ocjena 
		// _ECTS 
		// _naziv 

		// int* pok; 
		// cout << pok; -> adresa
		// cout << *pok; -> vrijednost

		// int pok;
		// cout << pok; -> vrijednost
		// cout << &pok; -> adresa


		if (this != &predmet) {

			delete _ECTS;

			delete[] _naziv;


			_ocjena = predmet._ocjena;

			_ECTS = new int(*predmet._ECTS);

			int vel = strlen(predmet._naziv) + 1;

			_naziv = new char[vel];

			strcpy_s(_naziv, vel, predmet._naziv);
		}

		return *this;

	}



	// 3. destruktor

	// TILDA -> alt gr + 1 ~~~~~

	~Predmet() {

		cout << "DESTRUTKOR SE POZVAO" << endl;

		delete _ECTS;

		_ECTS = nullptr;

		delete[] _naziv;

		_naziv = nullptr;

	}



	// 2. ispis

	void Ispis() {

		cout << "Naziv predmeta: " << _naziv << endl;
		cout << "Ocjena predmeta: " << _ocjena << endl;
		cout << "ECTS predmeta: " << *_ECTS << endl;

	}



};
