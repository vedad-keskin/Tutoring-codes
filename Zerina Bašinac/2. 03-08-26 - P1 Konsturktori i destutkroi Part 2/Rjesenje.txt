// POKAZIVACI

// 1. ALOKACIJA ( new ) 

// - ako nam treba jedna vrijednost -> ( ) 

// - ako nam treba vise vrijednosti -> [ ] 

// 2. DEREFERECIRANJE ( * ) 

// - ako nam treba jedna vrijednost -> ( ) -> mora ici * -> _ECTS = 0000026C36A58740 , *_ECTS = 0

// - ako nam treba vise vrijednosti -> [ ] -> ne smije ici * -> _naziv = Nema predmeta , *_naziv = N 

// 3. DESTRUKTOR ILI DEALOKACIJA ( delete , nullptr ) 

// - ako nam treba jedna vrijednost -> ( ) -> delete _ECTS; 

// - ako nam treba vise vrijednosti -> [ ] -> delete[] _naziv;


// this -> 0x213213 , *this -> _naziv , _ECTS , _ocjena .. 
class Predmet {
public:			  // dft. constr.  //
	char* _naziv; // Nema predmeta // nullptr
	//char _naziv[50];

	int _ocjena;  //      5		   //
	int* _ECTS;	  //      0		   //


	// 1. dft. constr.

	Predmet() {

		_ocjena = 5;

		_ECTS = new int(0);

		//char dftNaziv[] = "Nema predmeta"; // 14
		//string dftNaziv = "Nema predmeta";
		const char* dftNaziv = "Nema predmeta"; // string literal -> read-only mem. 

		int vel = strlen(dftNaziv) + 1; // 13 sl. + 1 = 14 sl.

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, dftNaziv);


	}

	// 4. 	Predmet predmet1("Programiranje II", 9, 7); // <--- user-def. constr.

	Predmet(const char* naziv, int ocjena, int ects) {

		_ocjena = ocjena;

		_ECTS = new int(ects);

		int vel = strlen(naziv) + 1;

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, naziv);


	}

	// 5. copy constr.
	//Predmet predmetKopija2(predmet2); // shallow copy -> deep copy -> copy constr.

	// const -> ne moze mijnejati
	// bez & -> napravi kopiju
	// sa & -> proslijeid original
	// & + const -> proslijeid original i zastiti taj origianl 



	Predmet(const Predmet& predmet) { // napraviti kopija tog predmeta2 -> posalje original 

		//predmet._ocjena = 10;

		_ocjena = predmet._ocjena;

		_ECTS = new int(*predmet._ECTS);

		int vel = strlen(predmet._naziv) + 1;

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, predmet._naziv);

	}



	// 6. op = 

	//predmetOperatorDodjele = predmet2; // shallow copy -> deep copy -> operator = -> operator dodjele

	Predmet& operator = (const Predmet& predmet) {

		// adresa L != adrese R 
		if (this != &predmet) {

			delete _ECTS;

			delete[] _naziv;


			_ocjena = predmet._ocjena;

			_ECTS = new int(*predmet._ECTS);

			int vel = strlen(predmet._naziv) + 1;

			_naziv = new char[vel];

			strcpy_s(_naziv, vel, predmet._naziv);

		}


		return *this;

	}











	// 3. destruktor ili dealokacija 

	// TILDA -> alt gr + 1 ~~~~~~

	~Predmet() {

		delete _ECTS;

		_ECTS = nullptr;

		delete[] _naziv;

		_naziv = nullptr;


	}




	// 2. ispis

	void Ispis() {


		cout << "Naziv predmeta: " << _naziv << endl;
		cout << "Ocjena predmeta: " << _ocjena << endl;
		cout << "ECTS predmeta: " << *_ECTS << endl;


	}



};



class Student {
	char* _imePrezime;


	Predmet* _polozeni; // ( ) ili [ ] <-- 100%
	int _trenutno;		// koliko trenutno imamo pol. predmeta
	int _max;			// koliko maksimalno mozemo imati pol. predmeta
public:


	Student() {

		const char* dftNaziv = "Nema imena i prezimena";

		int vel = strlen(dftNaziv) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, dftNaziv);

		_trenutno = 0;
		_max = 3;

		_polozeni = new Predmet[_max];


	}

	Student(const char* imeprezime, int max) {

		int vel = strlen(imeprezime) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, imeprezime);

		_trenutno = 0;
		_max = max;

		_polozeni = new Predmet[_max];





	}



	Student(const Student& obj) {

		int vel = strlen(obj._imePrezime) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, obj._imePrezime);

		_trenutno = obj._trenutno;
		_max = obj._max;

		_polozeni = new Predmet[_max];


		for (int i = 0; i < _trenutno; i++) // 0 1 2 
		{
			_polozeni[i] = obj._polozeni[i];
		}

	}


	Student& operator = (const Student& obj) {
		if (this != &obj) {


			delete[] _imePrezime;

			delete[] _polozeni;

			int vel = strlen(obj._imePrezime) + 1;
			_imePrezime = new char[vel];
			strcpy_s(_imePrezime, vel, obj._imePrezime);

			_trenutno = obj._trenutno;
			_max = obj._max;

			_polozeni = new Predmet[_max];


			for (int i = 0; i < _trenutno; i++) // 0 1 2 
			{
				_polozeni[i] = obj._polozeni[i];
			}

		}
		return *this;
	}


	~Student() {

		delete[] _imePrezime;

		_imePrezime = nullptr;

		delete[] _polozeni;

		_polozeni = nullptr;

	}

	void Ispis() {

		cout << "Ime i prezime studenta: " << _imePrezime << endl;


		for (int i = 0; i < _trenutno; i++) // 0 1 
		{
			cout << endl;
			_polozeni[i].Ispis();
		}


	}







	//student1.AddPolozeniPredmet(predmet1); // MATII

	void AddPolozeniPredmet(Predmet predmet) {



		// Voditi računa da se ne može prekoračiti kapacitet niza

		if (_max == _trenutno) {

			cout << "Student ne moze poloziti vise predmeta" << endl;
			return;

		}

		// Voditi računa da se ne moze dodavati isti predmet (istog naziva)		        


		for (int i = 0; i < _trenutno; i++) // 0 1 
		{

			// MATII == PRII   ,, i = 0
			// MATII == MATII  ,, i = 1 --> return ; 

			//           char* == char* -> NE RADI 
			//      strcmp ( char* , char* ) == 0 -> RADI

			if (strcmp( predmet._naziv , _polozeni[i]._naziv ) == 0) {


				cout << "Student ne moze poloziti isti predmet" << endl;
				return;

			}


		}




		// Dodaje novi polozeni predmet u niz

		//_polozeni[0] = predmet; // op = od klase Predmeta

		_polozeni[_trenutno] = predmet; // pol[0] = PRII , pol[1] = MATII , pol[2] = WRD
		 
		_trenutno++; // 0 --> 1 --> 2 --> 3


	}

};
