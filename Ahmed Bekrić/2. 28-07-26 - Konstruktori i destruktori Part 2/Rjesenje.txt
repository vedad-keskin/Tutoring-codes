// POKAZIVACI

// 1. ALOKACIJA ( new ) 

// ( ) --> jedno mem. mjesto

// [ ] --> vise mem. mjesta 

// 2. DEREFERENCIRANJE ( * ) 

// ( ) --> jedno mem. mjesto --> mora deref. *_ECTS = 0 , _ECTS = 00000198BCDD8200

// [ ] --> vise mem. mjesta --> ne smije deref. *_naziv = N , _naziv = Nema predmeta

// 3. DESTRUKTOR / DEALOAKCIJA ( delete , nullptr ) 

// ( ) --> jedno mem. mjesto --> delete _ECTS; 

// [ ] --> vise mem. mjesta -->  delete[] _naziv;



class Predmet {
public:			  // dft. constr.  //
	char* _naziv; // Nema predmeta // nullptr
	//char _naziv[50];

	int _ocjena;  //     5		   //
	int* _ECTS;	  //     0 		   //


	// 1. dft. const

	Predmet() {

		_ocjena = 5;

		_ECTS = new int(0);

		//char dftNaziv[] = "Nema predmeta"; // 14 
		//string dftNaziv = "Nema predmeta";
		const char* dftNaziv = "Nema predmeta"; // string literal -> read only mem. 

		int vel = strlen(dftNaziv) + 1; // 13 + 1 = 14 

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, dftNaziv);


	}

	// 4. user-def constr.
	//Predmet predmet1("Programiranje II", 9, 7); // <-- user-def constr.

	Predmet(const char* naziv, int ocjena, int ects) {

		_ocjena = ocjena;

		_ECTS = new int(ects);

		int vel = strlen(naziv) + 1;

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, naziv);


	}

	// 5. copy constr. 
	//Predmet predmetKopija2(predmet2);  // shallow copy --> deep copy --> copy constr.

	// bez & -> pravimo kopiju
	// sa & -> saljemo original
	// sa const + & -> stitimo original

	Predmet(const Predmet& predmet) { // napravi kopija tog predmet1 -> posalji original 

		//predmet._ocjena = 10;

		_ocjena = predmet._ocjena;

		_ECTS = new int(*predmet._ECTS);

		int vel = strlen(predmet._naziv) + 1;

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, predmet._naziv);

	}

	// 6. op = 
	//predmetOperatorDodjele = predmet1;  // shallow copy --> deep copy --> operator = 

	Predmet& operator = (const Predmet& predmet) {

		if (this != &predmet) {

			delete _ECTS;

			delete[] _naziv;



			_ocjena = predmet._ocjena;

			_ECTS = new int(*predmet._ECTS);

			int vel = strlen(predmet._naziv) + 1;

			_naziv = new char[vel];

			strcpy_s(_naziv, vel, predmet._naziv);

		}





		// this -> adresa tog glavnog objekta
		// *this -> ref. glavni obj.

		return *this;

	}





	// 3. desturktor / dealokacija 

	// TILDA -> alt gr + 1 -> ~~~~~~

	~Predmet() {

		//cout << "DESTUKTOR SE POZVAO" << endl;

		delete _ECTS;

		_ECTS = nullptr;

		delete[] _naziv;

		_naziv = nullptr;

	}





	// 2. ispis

	void Ispis() {

		cout << "Naziv predmeta: " << _naziv << endl;
		cout << "Ocjena predmeta: " << _ocjena << endl;
		cout << "ECTS predmeta: " << *_ECTS << endl;

	}



};



class Student {
	char* _imePrezime;

	Predmet* _polozeni; // ( ) ili [ ] <- 100% .. 
	int _trenutno;		// koliko trenutno imamo pohranjenih predmeta
	int _max;			// koliko maximalno mozem imati poh. predmeta
public:


	Student() {

		const char* dftIP = "Nema imena i prezimena";

		int vel = strlen(dftIP) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, dftIP);

		_trenutno = 0;

		_max = 3;

		_polozeni = new Predmet[_max];

	}

	Student(const char* imeprezime, int max) {

		int vel = strlen(imeprezime) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, imeprezime);

		_trenutno = 0;

		_max = max;

		_polozeni = new Predmet[_max];


	}

	Student(const Student& obj) {

		int vel = strlen(obj._imePrezime) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, obj._imePrezime);

		_trenutno = obj._trenutno;
		_max = obj._max;

		_polozeni = new Predmet[_max];

		for (int i = 0; i < _trenutno; i++) // 0 1 2 
		{
			_polozeni[i] = obj._polozeni[i];
		}

	}

	Student& operator = (const Student& obj) {
		if (this != &obj) {


			delete[] _imePrezime; // ctrl + d

			delete[] _polozeni;



			int vel = strlen(obj._imePrezime) + 1;
			_imePrezime = new char[vel];
			strcpy_s(_imePrezime, vel, obj._imePrezime);

			_trenutno = obj._trenutno;
			_max = obj._max;

			_polozeni = new Predmet[_max];

			for (int i = 0; i < _trenutno; i++) // 0 1 2 
			{
				_polozeni[i] = obj._polozeni[i];
			}

		}
		return *this;
	}



	~Student() {

		delete[] _imePrezime; // ctrl + d

		_imePrezime = nullptr;

		delete[] _polozeni;
	
		_polozeni = nullptr;

	}


	void Ispis() {

		cout << "Ime i prezime: " << _imePrezime << endl;

		cout << "POLOZENI PREDMETI STUDENTA: " << endl;

		for (int i = 0; i < _trenutno; i++) // 0 1 2 
		{

			cout << endl;
			_polozeni[i].Ispis(); // pol[0] , pol[1] , pol[2] .. 

		}

		//_polozeni[0].Ispis();


	}












	//student.AddPolozeniPredmet(predmet1); // PRII  ,  MATII 
	void AddPolozeniPredmet(Predmet predmet) {


		// Voditi računa da se ne može prekoračiti kapacitet niza

		if (_max == _trenutno) {

			cout << "Prekoracen niz" << endl;
			return;

		}

		// Voditi računa da se ne moze dodavati isti predmet (istog naziva)		        

		for (int i = 0; i < _trenutno; i++)
		{
			// pol[0] = PRII        == MATII
			
			// pol[1] = MATII       == MATII

			//                char* == char* 
			//          strcmp( char* , char* ) == 0 


			if (strcmp(_polozeni[i]._naziv , predmet._naziv ) == 0 )  {

				cout << "Duplikat" << endl;
				return;

			}

		}






		// Dodaje novi polozeni predmet u niz

		_polozeni[_trenutno] = predmet; // op = Predmet -> pol[0] = PRII , pol[1] = MATII

		_trenutno++; // 0 --> 1 --> 2 


	}

};
