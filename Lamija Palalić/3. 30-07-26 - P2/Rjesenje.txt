char* GetNizKaraktera(const char* tekst) { // AlocirajNiz , AlocirajTekst

	int vel = strlen(tekst) + 1;

	char* novi = new char[vel];

	strcpy_s(novi, vel, tekst);

	return novi;

}


// BAZNA KLASA -> KLASA KOJA SE NASLIJEDJUJE 

class Osoba {
protected: // 1/3 RAZLIKA
	char* _imePrezime;
	char* _JMBG;
public:


	// user-def + dft 


	//               "Denis Music",              "1232132132");
	Osoba(const char* imeprezime = "---", const char* jmbg = "---") {


		_imePrezime = GetNizKaraktera(imeprezime);

		_JMBG = GetNizKaraktera(jmbg);

	}

	Osoba(const Osoba& obj) {

		_imePrezime = GetNizKaraktera(obj._imePrezime);

		_JMBG = GetNizKaraktera(obj._JMBG);

	}

	Osoba& operator = (const Osoba& obj) {
		if (this != &obj) {

			delete[] _imePrezime;

			delete[] _JMBG;

			_imePrezime = GetNizKaraktera(obj._imePrezime);

			_JMBG = GetNizKaraktera(obj._JMBG);

		}
		return *this;
	}

	virtual ~Osoba() { // 3/3 RAZLIKA 

		cout << "POZVAO SE BAZNI DESTRUKTOR" << endl;


		delete[] _imePrezime;
		_imePrezime = nullptr;
		delete[] _JMBG;
		_JMBG = nullptr;

	}


	virtual void Test() { // virtual se koristi samo u baznim klasama i kaze nam da neka metoda se moze ali i ne mora promijeniti

		cout << "POCETNA IMPLEMENTACIJA METODE TEST() " << endl;

	}


	virtual void Info() = 0; // pure virt. func. 2/3 RAZLIKA

	// 1. citava klasa Osoba postaje apstraktna -> nasa klasa vise nije obicna klasa nego sada ona postaje samo neki pomocni mehanizam da lakse pravimo izvedne klase

	// 2. svaka izvedena klasa mora imati override/promjenu te metode


	void Ispis() {

		cout << _imePrezime << " [" << _JMBG << "]" << endl;

	}


	//cout << osoba0;

	friend ostream& operator << (ostream& COUT , const Osoba& obj) {

		COUT << obj._imePrezime << " [" << obj._JMBG << "]" << endl;

		return COUT;

	}




};


// IZVEDENA -> klasa koja naslijedi baznu klasu 

class Profesor : public Osoba { // 1/5 KORAKA
public:
	char* _titula;


	// Profesor profesor1("dr. sc.", "Denis Music", "1232132132");

	Profesor(const char* titula = "prof.", const char* imeprezime = "---", const char* jmbg = "---") : Osoba(imeprezime, jmbg)  { // 2/5 KORAKA , 

		_titula = GetNizKaraktera(titula);

	}

	Profesor(const Profesor& obj) : Osoba(obj) { // 3/5 KORAK

		_titula = GetNizKaraktera(obj._titula);


	}

	Profesor& operator = (const Profesor& obj)  {
		if (this != &obj) {

			delete[] _titula;

			Osoba::operator=(obj); // 4/5 KORAK

			_titula = GetNizKaraktera(obj._titula);

		}
		return *this;
	}

	void Test() override {

		cout << "NAPREDNA IMPLEMENTACIJA METODE TEST() " << endl;

	}

	void Info() override { // 5/5 KORAK

		cout << *this;

	}

	~Profesor() {

		cout << "POZVAO SE IZVEDENI DESTRUKTOR" << endl;

		delete[] _titula;

		_titula = nullptr;

	}

	void Ispis() {

		cout << _titula << " " << _imePrezime << " [" << _JMBG << "]" << endl;

	}

	friend ostream& operator << (ostream& COUT, const Profesor& obj) {

		COUT << obj._titula << " " << obj._imePrezime << " [" << obj._JMBG << "]" << endl;

		return COUT;

	}



};
