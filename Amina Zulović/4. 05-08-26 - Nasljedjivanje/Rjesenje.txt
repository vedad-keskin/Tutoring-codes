// GetNizKaraktera / AlocirajNiz / AlocirajTekst

char* GetNizKaraktera(const char* tekst) {

	int vel = strlen(tekst) + 1;

	char* novi = new char[vel];

	strcpy_s(novi, vel, tekst);

	return novi;

}



class Osoba { // BAZNA KLASA
protected: // 1/3 RAZLIKA
	char* _imePrezime;
	char* _JMBG;
public:



	Osoba(const char* imeprezime = "---", const char* jmbg = "---") {

		_imePrezime = GetNizKaraktera(imeprezime);
		_JMBG = GetNizKaraktera(jmbg);

	}

	Osoba(const Osoba& obj) {

		_imePrezime = GetNizKaraktera(obj._imePrezime);
		_JMBG = GetNizKaraktera(obj._JMBG);

	}

	Osoba& operator = (const Osoba& obj) {
		if (this != &obj) {
			delete[] _imePrezime;
			delete[] _JMBG;

			_imePrezime = GetNizKaraktera(obj._imePrezime);
			_JMBG = GetNizKaraktera(obj._JMBG);
		}
		return *this;
	}

	virtual ~Osoba() { // 3/3 RAZLIKA

		cout << "POZVAO SE BAZNI DESTRUTKOR" << endl;


		delete[] _imePrezime;
		_imePrezime = nullptr;
		delete[] _JMBG;
		_JMBG = nullptr;

	}

	virtual void Test() { // OPC. 


		cout << "VER.1 NEKE TEST() METODE" << endl;

	}



	virtual void Info() = 0; // 2/3 RAZLIKE -> pure virt. func. -> MORA OVERRIDE  

	// Osoba -> postaje asptraktna klasa




	void Ispis() {

		cout << _imePrezime << " [" << _JMBG << "]" << endl;

	}

	//cout << osoba0;

	friend ostream& operator << (ostream& COUT, const Osoba& obj) {


		COUT << obj._imePrezime << " [" << obj._JMBG << "]" << endl;

		return COUT;

	}




};

// IZVEDENA KLASA


class Profesor : public Osoba {  // 1/5 KORAK
public:
	char* _titula;


	Profesor(const char* titula = "prof.", const char* imeprezime = "---", const char* jmbg = "---") : Osoba(imeprezime, jmbg) { // 2/5 KORAK


		_titula = GetNizKaraktera(titula);


	}

	Profesor(const Profesor& obj) : Osoba(obj) { // 3/5 KORAK


		_titula = GetNizKaraktera(obj._titula);

	}

	Profesor& operator = (const Profesor& obj) {
		if (this != &obj) {


			delete[] _titula;


			Osoba::operator=(obj); // 4/5 KORAK


			_titula = GetNizKaraktera(obj._titula);

		}
		return *this;
	}


	void Test() override {


		cout << "VER.2 NEKE TEST() METODE" << endl;

	}


	void Info() override { // 5/5 KORAK

		cout << *this;

	}


	~Profesor() {

		cout << "POZVAO SE IZVEDENI DESTRUTKOR" << endl;

		delete[] _titula;

		_titula = nullptr;

	}

	//cout << osoba0;

	friend ostream& operator << (ostream& COUT, const Profesor& obj) {


		COUT << obj._titula<<" "<<  obj._imePrezime << " [" << obj._JMBG << "]" << endl;

		return COUT;

	}



};
