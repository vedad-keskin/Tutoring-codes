// POKAZIVACI 

// 1. ALOKACIJA ( new ) 

// ako nam treba jedan mem. proostor -> ( ) 

// ako nam treba vise mem. proostor -> [ ] 

// 2. DEREFERENCIRANJE ( * ) 

// ako nam treba jedan mem. proostor -> ( ) -> * je obavezna -> _ECTS = 000001FD144886C0 , *_ECTS = 0

// ako nam treba vise mem. proostor -> [ ] -> * ne smije pisati -> _naziv = Nema predmeta , *_naziv = N 

// 3. DESTRUTKOR / DEALOKACIJA

// ako nam treba jedan mem. proostor -> ( ) -> delete _ECTS;

// ako nam treba vise mem. proostor -> [ ] -> delete[] _naziv;



// this
class Predmet {
public:			  // dft. constr.  //
	char* _naziv; // Nema predmeta // nullptr
	//char _naziv[50];

	int _ocjena;  //      5		   //
	int* _ECTS;	  //      0		   //


	// 1. dft. constr.

	Predmet() {

		_ocjena = 5;

		_ECTS = new int(0); // 1231312

		//char dftNaziv[] = "Nema predmeta"; // 14 slova
		//string dftNaziv = "Nema predmeta";
		const char* dftNaziv = "Nema predmeta"; // string literal -> read only mem. 

		int vel = strlen(dftNaziv) + 1; // 13 sl. + 1 = 14 

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, dftNaziv);


	}

	// 4. user-def constr.
	//Predmet predmet1("Programiranje II", 9, 7); // <-- user-def constr.

	Predmet(const char* naziv, int ocjena, int ects) {

		_ocjena = ocjena;

		_ECTS = new int(ects);

		int vel = strlen(naziv) + 1;

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, naziv);

	}

	// 5. copy constr.
	//Predmet predmetKopija2(predmet2);  // shallow copy --> deep copy -> copy constr.

	// bez & -> pravimo kopiju
	// sa & -> saljemo original
	// const -> nesto je konstantno -> ne moze se mijenajti



	Predmet(const Predmet& predmet) { // napraviti kopija tog predmeta1 -> poslaji original predmeta1

		//predmet._ocjena = 10;

		_ocjena = predmet._ocjena;

		_ECTS = new int(*predmet._ECTS);

		int vel = strlen(predmet._naziv) + 1;

		_naziv = new char[vel];

		strcpy_s(_naziv, vel, predmet._naziv);

	}

	// 6. op = 
	//predmetOperatorDodjele = predmet1; // shallow copy --> deep copy --> op = 

	Predmet& operator = (const Predmet& predmet) {


		// this -> 0x2132132 , *this 
		// _ocjena = 9
		// _ECTS = 7
		// _naziv = Prog.. 


		if (this != &predmet) {

			delete _ECTS;

			delete[] _naziv;


			_ocjena = predmet._ocjena;

			_ECTS = new int(*predmet._ECTS);

			int vel = strlen(predmet._naziv) + 1;

			_naziv = new char[vel];

			strcpy_s(_naziv, vel, predmet._naziv);

		}

		return *this;

	}




	// 3. destruktor
	// tilda -> alt gr + 1 ~~~~~~

	~Predmet() {

		//cout << "DESTRUKTOR SE POZVAO" << endl;

		delete _ECTS;

		_ECTS = nullptr;

		delete[] _naziv;

		_naziv = nullptr;

	}


	// 2. ispis

	void Ispis() {

		cout << "Naziv predmeta: " << _naziv << endl;
		cout << "Ocjena predmeta: " << _ocjena << endl;
		cout << "ECTS predmeta: " << *_ECTS << endl;

	}



};


class Student {
	char* _imePrezime;

	Predmet* _polozeni; // ( ) ili [ ] <-- 100% 
	int _trenutno;		// koliko imamo trenutno pol. predmeta
	int _max;			// koliko max mozemo imati pol. predmeta
public:



	Student() {

		const char* dftIP = "Nema imena i prezimena";
		int vel = strlen(dftIP) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, dftIP);

		_trenutno = 0;
		_max = 3;

		_polozeni = new Predmet[_max];


	}

	Student(const char* imeprezime, int max) {

		int vel = strlen(imeprezime) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, imeprezime);

		_trenutno = 0;

		_max = max;

		_polozeni = new Predmet[_max];



	}


	Student(const Student& obj) {

		int vel = strlen(obj._imePrezime) + 1;
		_imePrezime = new char[vel];
		strcpy_s(_imePrezime, vel, obj._imePrezime);

		_trenutno = obj._trenutno;
		_max = obj._max;

		_polozeni = new Predmet[_max];

		for (int i = 0; i < _trenutno; i++)
		{
			_polozeni[i] = obj._polozeni[i];
		}

	}


	Student& operator = (const Student& obj) {
		if (this != &obj) {
			


			delete[] _imePrezime;

			delete[] _polozeni;


			int vel = strlen(obj._imePrezime) + 1;
			_imePrezime = new char[vel];
			strcpy_s(_imePrezime, vel, obj._imePrezime);

			_trenutno = obj._trenutno;
			_max = obj._max;

			_polozeni = new Predmet[_max];

			for (int i = 0; i < _trenutno; i++)
			{
				_polozeni[i] = obj._polozeni[i];
			}

		}
		return *this;
	}


	~Student() {

		delete[] _imePrezime;

		_imePrezime = nullptr;

		delete[] _polozeni;

		_polozeni = nullptr; // ctrl + d


		// ctrl + c
		// ctrl + v

	}


	void Ispis() {


		cout << "Ime i prezime studenta: " << _imePrezime << endl;

		for (int i = 0; i < _trenutno; i++) // 0 1 
		{
			cout << endl;
			_polozeni[i].Ispis(); // pol[0] , pol[1] 

		}


		//_polozeni[0].Ispis();

	}





	//student1.AddPolozeniPredmet(predmet1); // PRII --> MATII
	void AddPolozeniPredmet(Predmet predmet) {


		// Voditi računa da se ne može prekoračiti kapacitet niza

		if (_trenutno == _max) {

			cout << "Prekoracen je kapacitet niza" << endl;
			return;

		}


		// Voditi računa da se ne moze dodavati isti predmet (istog naziva)	
		
		for (int i = 0; i < _trenutno; i++)
		{

			// _polozeni[i]._naziv -> _polozeni[0]._naziv -> PRII
			// _polozeni[i]._naziv -> _polozeni[1]._naziv -> MATII

			// predmet._naziv -> MATII 

			// _polozeni[0]._naziv -> PRII == predmet._naziv -> MATII 
			// _polozeni[1]._naziv -> MATII == predmet._naziv -> MATII 


			//                char* == char*
			//            strcmp ( char* , char* ) == 0 

			if (strcmp( _polozeni[i]._naziv , predmet._naziv ) == 0) {

				cout << "Duplikat" << endl;
				return;

			}

		}




		// Dodaje novi polozeni predmet u niz
		//_polozeni[0] = predmet; // op = od klase Predmet

		_polozeni[_trenutno] = predmet; // pol[0] = PRII , pol[1] = MATII

		_trenutno++; // 0 --> 1 --> 2


	}

};
