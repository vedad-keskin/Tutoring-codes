char* AlocirajTekst(const char* tekst) { // AlocitajNiz , GetNizKaraktera
	if (!tekst) return nullptr;


	size_t vel = strlen(tekst) + 1;
	char* temp = new char[vel];
	strcpy_s(temp, vel, tekst);
	return temp;


}


//funkcija generise clanski broj na osnovu imena i prezimena, rednog broja i trenutne godine.
//clanski broj je u formatu GGGG/IN-BBB, gdje GGGG predstavlja trenutnu godinu, IN inicijale,
//a BBB redni broj clana popunjen nulama na tri mjesta.
//funkciju koristiti u konstruktoru klase ClanBiblioteke za inicijalizaciju atributa _clanskiBroj.

string GenerisiClanskiBroj(const char* imePrezime, int redniBroj) {

	string clanskiBroj;

	// GGGG/IN-BBB

	//clanskiBroj += "2026";

	time_t utcVrijeme = time(0);

	tm lokalnoVrijeme;

	// int* pok; -> 5

	// cout << *pok; -> 5
	// cout << pok; -> 0x2132131

	// int broj; -> 5

	// cout << broj; -> 5
	// cout << &broj; -> 0x213123


	localtime_s(&lokalnoVrijeme, &utcVrijeme);

	int godina = lokalnoVrijeme.tm_year + 1900; // 126 + 1900 = 2026

	clanskiBroj += to_string(godina);






	// 2. /IN

	clanskiBroj += "/";

	char prvoSlovoImena = toupper(imePrezime[0]);

	string ip = imePrezime;

	char prvoSlovoPrezimena = toupper(imePrezime[ip.rfind(' ') + 1]);


	clanskiBroj += prvoSlovoImena;
	clanskiBroj += prvoSlovoPrezimena;




	// 3. -BBB

	clanskiBroj += "-";

	//if(redniBroj < 10 ){
	//
	//	clanskiBroj += "00";

	//}
	//else if (redniBroj < 100) {

	//	clanskiBroj += "0";

	//}

	//clanskiBroj += to_string(redniBroj);

	stringstream ss;

	ss << setfill('0') << setw(3) << redniBroj;

	clanskiBroj += ss.str();



	return clanskiBroj;

}





//ValidirajClanskiBroj koristeci regex provjerava format definisan u prethodnom dijelu zadatka.
//if (ValidirajClanskiBroj("2026/AB-003"))

// GGGG/IN-BBB

// 1. RANGE [ ] -> [A-Z]{3} , [a-z] , [0-9] , [A-Za-z0-9] , [- ]

// 2. VELICINA { } -> {3} , {3,6} , {,5} , {5,} 

// 3. TACAN TEKST ( ) -> (edu) , (com|ba) -> | u regexu , || u c++ 


bool ValidirajClanskiBroj(const string& clanskiBroj) {

	regex uzorak("[0-9]{4}[/][A-Z]{2}[-][0-9]{3}");

	return regex_match(clanskiBroj, uzorak);

}

template<class T1, class T2, int max>
class Kolekcija {
	T1* _prvi;
	//T1* _prvi[max];
	//T1 _prvi[max];



	T2* _drugi;
	int _trenutno;
public:
	Kolekcija() : _prvi(nullptr), _drugi(nullptr), _trenutno(0) {
	
		//_trenutno = 0;
		//_prvi = nullptr;
		//_drugi = nullptr;

		_prvi = new T1[max];
		_drugi = new T2[max];

	}

	Kolekcija(const Kolekcija& obj) {

		_trenutno = obj._trenutno;

		_prvi = new T1[max];
		_drugi = new T2[max];

		for (int i = 0; i < _trenutno; i++)
		{
			_prvi[i] = obj._prvi[i];
			_drugi[i] = obj._drugi[i];
		}

	}

	Kolekcija& operator = (const Kolekcija& obj) {
		if (this != &obj) {

			delete[] _prvi;
			delete[] _drugi;


			_trenutno = obj._trenutno;

			_prvi = new T1[max];
			_drugi = new T2[max];

			for (int i = 0; i < _trenutno; i++)
			{
				_prvi[i] = obj._prvi[i];
				_drugi[i] = obj._drugi[i];
			}
		}
		return *this;
	}


	void Dodaj(T1 el1, T2 el2) {


		_prvi[_trenutno] = el1;
		_drugi[_trenutno] = el2;

		_trenutno++;

	}


	//DodajNaPoziciju dodaje novi par na lokaciju definisanu prvim parametrom. metoda vraca trenutno stanje kolekcije.
	//u slucaju popunjene kolekcije ili neispravne lokacije potrebno je baciti izuzetak (za potrebe testiranja mozete dodati try catch blok).
	//Kolekcija<int, string, 20> prosireniInventar = inventar.DodajNaPoziciju(2, 99, "Posebno izdanje");

	Kolekcija DodajNaPoziciju(int pozicija, T1 el1, T2 el2) {

		if (_trenutno == max || pozicija < 0 || pozicija >= _trenutno) {

			throw exception("Neispravan unos");

		}


		Kolekcija prosireniInventar;


		for (int i = 0; i < _trenutno; i++) // 0 1 2 3 4 5 6 7 ...
		{

			if (pozicija == i) { // 2 == 3

				prosireniInventar.Dodaj(el1, el2);

			}

			prosireniInventar.Dodaj(_prvi[i], _drugi[i]);


		}

		return prosireniInventar;


	}


	//UkloniRaspon uklanja broj elemenata definisan drugim parametrom, pocevsi od lokacije
	//definisane prvim parametrom (ukljucujuci tu lokaciju). metoda vraca pokazivac na novu kolekciju sa uklonjenim
	//elementima, a pozivalac je odgovoran za njenu dealokaciju.
	//Kolekcija<int, string, 20>* uklonjeneKnjige = prosireniInventar.UkloniRaspon(3, 2);

	Kolekcija* UkloniRaspon(int lokacijaOd, int brojElemenata) {


		if (lokacijaOd < 0 || lokacijaOd + brojElemenata >= _trenutno) {


			throw exception("Neispravan unos");

		}


		Kolekcija* uklonjeneKnjige = new Kolekcija();

		Kolekcija preostaleKnjige;


		for (int i = 0; i < _trenutno; i++)
		{
			//    >= 3 ... < 5
			if (i >= lokacijaOd && i < lokacijaOd + brojElemenata) {

				uklonjeneKnjige->Dodaj(_prvi[i], _drugi[i]);

			}
			else {

				preostaleKnjige.Dodaj(_prvi[i], _drugi[i]);

			}

		}



		*this = preostaleKnjige;

		return uklonjeneKnjige;

	}






	int GetTrenutno() const { return _trenutno; }
	int GetTrenutno() { return _trenutno; }

	T1& GetPrvi(int indeks) { return _prvi[indeks]; }
	T2& GetDrugi(int indeks) { return _drugi[indeks]; }
	T1& operator[](int indeks) { return _prvi[indeks]; }

	T1& GetPrvi(int indeks) const { return _prvi[indeks]; }
	T2& GetDrugi(int indeks) const { return _drugi[indeks]; }
	T1& operator[](int indeks) const { return _prvi[indeks]; }


	friend ostream& operator<<(ostream& COUT, const Kolekcija& obj) {
		for (int i = 0; i < obj.GetTrenutno(); i++)
			COUT << obj.GetPrvi(i) << " " << obj.GetDrugi(i) << endl;
		return COUT;
	}

	~Kolekcija() {
		delete[] _prvi; _prvi = nullptr;
		delete[] _drugi; _drugi = nullptr;
	}
};



class Datum { // DatumVrijeme, Vrijeme , Termin 
	int* _godina, * _mjesec, * _dan;
public:
	Datum(int dan = 1, int mjesec = 1, int godina = 2000) {

		_godina = new int(godina);
		_mjesec = new int(mjesec);
		_dan = new int(dan);

	}

	Datum(const Datum& obj) {

		_godina = new int(*obj._godina);
		_mjesec = new int(*obj._mjesec);
		_dan = new int(*obj._dan);


	}

	Datum& operator = (const Datum& obj) {
		if (this != &obj) {

			delete _godina; delete _mjesec; delete _dan;


			_godina = new int(*obj._godina);
			_mjesec = new int(*obj._mjesec);
			_dan = new int(*obj._dan);

		}
		return *this;
	}

	bool operator == (const Datum& obj) const {

		return *_dan == *obj._dan && *_mjesec == *obj._mjesec && *_godina == *obj._godina;

	}

	~Datum() {

		cout << "DESTRUTKOR SE POZVAO" << endl;

		delete _godina; delete _mjesec; delete _dan;

		_godina = nullptr; _mjesec = nullptr; _dan = nullptr;

	}


	friend ostream& operator<<(ostream& COUT, const Datum& obj) {

		COUT << setfill('0') << setw(2) << *obj._dan << "." << setw(2) << *obj._mjesec << "." << *obj._godina;


		return COUT;
	}

};

class Posudba {
	char* _naslov;
	Zanr _zanr;
	Datum _datumPosudbe;
	int _brojDana;
public:
	Posudba(const char* naslov, Zanr zanr, Datum datumPosudbe, int brojDana) : _zanr(zanr), _datumPosudbe(datumPosudbe), _brojDana(brojDana) {

		_naslov = AlocirajTekst(naslov);

	}

	Posudba(const Posudba& obj) {

		_naslov = AlocirajTekst(obj._naslov);
		_zanr = obj._zanr;
		_datumPosudbe = obj._datumPosudbe;
		_brojDana = obj._brojDana;

	}

	Posudba& operator = (const Posudba& obj) {
		if (this != &obj) {
			delete[] _naslov;

			_naslov = AlocirajTekst(obj._naslov);
			_zanr = obj._zanr;
			_datumPosudbe = obj._datumPosudbe;
			_brojDana = obj._brojDana;
		}
		return *this;
	}

	~Posudba() {
		delete[] _naslov;
		_naslov = nullptr;
	}


	const char* GetNaslov() const { return _naslov; }
	Zanr GetZanr() const { return _zanr; }
	Datum& GetDatumPosudbe() { return _datumPosudbe; }
	int GetBrojDana() const { return _brojDana; }

	//ToString metoda vraca podatke o posudbi u formatu prikazanom u nastavku.
	//voditi racuna o prikazu jednocifrenih vrijednosti datuma (npr. 5 -> 05).
	//cout << tvrdjava.ToString() << crt;
	//05.10.2026 Tvrdjava ROMAN 14 dana


	bool operator == (const Posudba& obj) const {

		return strcmp(_naslov, obj._naslov) == 0;

	}

	string ToString() const {

		stringstream ss;

		ss << *this;

		return ss.str();


	}


	friend ostream& operator<<(ostream& COUT, const Posudba& obj) {

		COUT << obj._datumPosudbe << " " << obj._naslov << " " << obj._zanr << " " << obj._brojDana << " dana";

		return COUT;
	}



};
